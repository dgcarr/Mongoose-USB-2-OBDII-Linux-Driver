cmake_minimum_required(VERSION 3.20)
project(mongoose_j2534 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(GNUInstallDirs)
option(MONGOOSE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(MONGOOSE_FUZZ "Build libFuzzer codec target (Clang required)" OFF)
if(WIN32)
  add_executable(mongoose-reference tools/windows_reference.c)
  target_include_directories(mongoose-reference PRIVATE include)
  return()
endif()
find_package(Threads REQUIRED)
find_path(LIBUSB_INCLUDE_DIR libusb.h PATH_SUFFIXES libusb-1.0 REQUIRED)
find_library(LIBUSB_LIBRARY NAMES usb-1.0 REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(MONGOOSE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(mongoose_core STATIC src/codec.cpp src/session.cpp src/replay.cpp)
set_target_properties(mongoose_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_include_directories(mongoose_core PUBLIC include src)
target_link_libraries(mongoose_core PUBLIC Threads::Threads)
add_library(mongoose_usb STATIC src/usb.cpp)
set_target_properties(mongoose_usb PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_include_directories(mongoose_usb SYSTEM PRIVATE ${LIBUSB_INCLUDE_DIR})
target_link_libraries(mongoose_usb PUBLIC mongoose_core ${LIBUSB_LIBRARY})
add_library(mongoose_j2534 SHARED src/j2534.cpp)
target_link_libraries(mongoose_j2534 PRIVATE mongoose_usb)
target_include_directories(mongoose_j2534 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(mongoose_j2534 PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0 CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_link_options(mongoose_j2534 PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
add_executable(mongoose-diag tools/diagnostic.cpp)
target_link_libraries(mongoose-diag PRIVATE mongoose_usb)
add_executable(mongoose-client tools/client.c)
target_link_libraries(mongoose-client PRIVATE mongoose_j2534)
if(BUILD_TESTING)
  add_executable(mongoose-tests tests/core_tests.cpp)
  target_link_libraries(mongoose-tests PRIVATE mongoose_core)
  add_test(NAME core COMMAND mongoose-tests)
  add_executable(mongoose-api-tests tests/api_tests.cpp src/j2534.cpp)
  target_link_libraries(mongoose-api-tests PRIVATE mongoose_core)
  add_test(NAME api COMMAND mongoose-api-tests)
  add_executable(mongoose-abi-test tests/abi.c)
  target_include_directories(mongoose-abi-test PRIVATE include)
  add_test(NAME c_abi COMMAND mongoose-abi-test)
  add_test(NAME symbols COMMAND ${CMAKE_COMMAND} -DLIBRARY=$<TARGET_FILE:mongoose_j2534> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/symbols.cmake)
  add_test(NAME captured_discovery COMMAND mongoose-diag --discover --replay ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/discovery.replay)
  add_test(NAME captured_open_close COMMAND mongoose-diag --open-close --replay ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/open-close.replay)
  add_test(NAME captured_inspect COMMAND mongoose-diag --inspect --replay ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/inspect.replay)
endif()
if(MONGOOSE_FUZZ)
  add_executable(mongoose-fuzz tests/fuzz_codec.cpp src/codec.cpp)
  target_include_directories(mongoose-fuzz PRIVATE src include)
  target_compile_options(mongoose-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(mongoose-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
install(TARGETS mongoose_j2534 mongoose-diag mongoose-client LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/mongoose/j2534.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/mongoose)
install(FILES packaging/60-mongoose-j2534.rules DESTINATION ${CMAKE_INSTALL_DATADIR}/mongoose-j2534)
install(FILES README.md docs/VALIDATION.md docs/CAPTURING.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_NAME mongoose-j2534)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
include(CPack)

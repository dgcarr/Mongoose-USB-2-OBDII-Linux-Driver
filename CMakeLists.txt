# The cdc_acm backend is the default and needs no external library, so libusb is
# optional: -DMONGOOSE_LIBUSB=OFF builds a tty-only driver that configures on a host
# without the headers at all.

  find_path(LIBUSB_INCLUDE_DIR libusb.h PATH_SUFFIXES libusb-1.0)
  find_library(LIBUSB_LIBRARY NAMES usb-1.0)

  set(MONGOOSE_HAVE_LIBUSB ON)

add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
# Distribution-standard hardening, on by default and off under the sanitizers (which have their own runtime
# checks and conflict with some of these). A build with no type set gets RelWithDebInfo, because _FORTIFY_SOURCE
# does nothing without optimization; packagers pass their own type and flags and these add to them.
option(MONGOOSE_HARDENING "Build with stack protector, FORTIFY_SOURCE, RELRO/BIND_NOW, PIE and CET" ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
if(MONGOOSE_HARDENING AND NOT MONGOOSE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  include(CheckCXXCompilerFlag)
  include(CheckPIESupported)
  check_pie_supported(LANGUAGES C CXX)
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  add_compile_options(-fstack-protector-strong -Wformat=2 -Wformat-security)
  # A distribution's own flags (CFLAGS/CXXFLAGS reach CMAKE_*_FLAGS) usually set FORTIFY_SOURCE already, and a
  # second, different value is a redefinition; only add ours when they have not.
  if(NOT CMAKE_C_FLAGS MATCHES "FORTIFY_SOURCE" AND NOT CMAKE_CXX_FLAGS MATCHES "FORTIFY_SOURCE")
    add_compile_options($<$<NOT:$<CONFIG:Debug>>:-D_FORTIFY_SOURCE=3>)
  endif()
  check_cxx_compiler_flag(-fstack-clash-protection MONGOOSE_HAS_STACK_CLASH)
  if(MONGOOSE_HAS_STACK_CLASH)
    add_compile_options(-fstack-clash-protection)
  endif()
  check_cxx_compiler_flag(-fcf-protection=full MONGOOSE_HAS_CET)
  if(MONGOOSE_HAS_CET)
    add_compile_options(-fcf-protection=full)
  endif()
  add_link_options(-Wl,-z,relro -Wl,-z,now -Wl,-z,noexecstack -Wl,--as-needed)
endif()
if(MONGOOSE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

  add_library(mongoose_usb STATIC src/usb.cpp)
  set_target_properties(mongoose_usb PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
  target_include_directories(mongoose_usb SYSTEM PRIVATE ${LIBUSB_INCLUDE_DIR})
  target_link_libraries(mongoose_usb PUBLIC mongoose_core ${LIBUSB_LIBRARY})
  target_link_libraries(mongoose_backends PUBLIC mongoose_usb)
  target_compile_definitions(mongoose_backends PRIVATE MONGOOSE_WITH_LIBUSB=1)

                              INSTALL_DESTINATION ${MONGOOSE_CMAKE_DIR})

                                 VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)

        DESTINATION ${MONGOOSE_CMAKE_DIR})

# udev only reads rules from its own directories, so this is not under the doc or data dir. The default is
# relative, so it follows --prefix and DESTDIR: lib/udev/rules.d under /usr (and /usr/local), where udev reads it.
# A packager with another layout passes -DMONGOOSE_UDEV_RULES_DIR=... (absolute or relative).
set(MONGOOSE_UDEV_RULES_DIR "lib/udev/rules.d" CACHE STRING "Where to install the udev rule")
install(FILES packaging/60-mongoose-j2534.rules DESTINATION ${MONGOOSE_UDEV_RULES_DIR})
# systemd unit: `systemctl start mongoose-socketcan@mongoose0` creates the interface and runs the bridge, so a user
# need not repeat the ip link commands after every boot. Relative by default, so --prefix and DESTDIR work.
set(MONGOOSE_SYSTEMD_UNIT_DIR "lib/systemd/system" CACHE STRING "Where to install the systemd unit")
set(BINDIR "${CMAKE_INSTALL_FULL_BINDIR}")
configure_file(packaging/mongoose-socketcan@.service ${CMAKE_CURRENT_BINARY_DIR}/mongoose-socketcan@.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/mongoose-socketcan@.service DESTINATION ${MONGOOSE_SYSTEMD_UNIT_DIR})

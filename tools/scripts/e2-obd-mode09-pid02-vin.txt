# E2: OBD-II mode 09 PID 02 (VIN). The response is MULTI-FRAME, so this forces
# the first-frame / flow-control / consecutive-frame exchange. The single most
# valuable capture for ISO15765 timing and adapter-side segmentation.
# NOTE: the response contains the vehicle VIN - redact before committing.
open dev
connect dev ISO15765 NONE 500000 chan
filter chan FLOW ISO15765 ISO15765_FRAME_PAD 0000ffff 000007e8 000007e0 fc
write chan ISO15765 ISO15765_FRAME_PAD 000007df0902 1000
read chan 16 3000
stopfilter chan fc
disconnect chan
close dev

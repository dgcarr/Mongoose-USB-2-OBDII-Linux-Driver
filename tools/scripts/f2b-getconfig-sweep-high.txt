# F2b: GET_CONFIG for the remaining standard IDs and the Drew Tech vendor IDs, on a CAN channel.
# Companion to f2 (which covers 0x01-0x20). Read only. Never touches 0xC002-0xC00A
# (NON_VOLATILE_STORE_*): reading them would be harmless but they are excluded so nobody edits this
# script into a SET by accident. Rejected IDs are findings; getconfig never aborts the script.
open dev
connect dev CAN NONE 500000 chan
getconfig chan 0x21,0x22,0x23,0x24,0x25,0x26,0x27,0x28
getconfig chan 0x29,0x2a,0x2b,0x2c,0x2d,0x2e,0x2f,0x30
getconfig chan 0x8001,0x8010,0x8011,0x8012
getconfig chan 0x10000001,0x10000005,0x10000006,0x10000007,0x10000008
disconnect chan
close dev
